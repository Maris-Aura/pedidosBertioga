  <script>
    // Inicializar Ícones Lucide
    lucide.createIcons();

    // Estado Simulado da Aplicação
    let currentStep = 1;
    let cart = [];
    let selectedItem = null;
    let orderStatus = 1; // 1: Aguardando, 2: Em Produção, 3: Saiu para Entrega, 4: Entregue
    let audioContext = null;
    let alarmInterval = null;

    // Alternar Abas (Cliente vs Admin)
    function switchView(view) {
      if (view === 'client') {
        document.getElementById('view-client').classList.remove('hidden');
        document.getElementById('view-admin').classList.add('hidden');
        document.getElementById('btn-tab-client').className = "px-3 py-1.5 rounded-lg bg-amber-500 text-slate-950 font-bold transition";
        document.getElementById('btn-tab-admin').className = "px-3 py-1.5 rounded-lg bg-slate-800 hover:bg-slate-700 text-white font-medium transition";
      } else {
        document.getElementById('view-client').classList.add('hidden');
        document.getElementById('view-admin').classList.remove('hidden');
        document.getElementById('btn-tab-client').className = "px-3 py-1.5 rounded-lg bg-slate-800 hover:bg-slate-700 text-white font-medium transition";
        document.getElementById('btn-tab-admin').className = "px-3 py-1.5 rounded-lg bg-amber-500 text-slate-950 font-bold transition";
      }
    }

    // Modal de Produto
    function openProductModal(title, price) {
      selectedItem = { title, price };
      document.getElementById('modal-item-title').innerText = title;
      document.getElementById('product-modal').classList.remove('hidden');
    }

    function closeProductModal() {
      document.getElementById('product-modal').classList.add('hidden');
    }

    function confirmAddToCart() {
      cart.push(selectedItem);
      closeProductModal();
      updateCartUI();
    }

    function updateCartUI() {
      if (cart.length > 0) {
        document.getElementById('cart-bar').classList.remove('hidden');
        document.getElementById('cart-count').innerText = `${cart.length} ${cart.length === 1 ? 'item' : 'itens'}`;
        const total = cart.reduce((sum, item) => sum + item.price, 0);
        document.getElementById('cart-total').innerText = `R$ ${total.toFixed(2).replace('.', ',')}`;
      }
    }
